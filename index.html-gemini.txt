    <script>
        const months = ["AĞUSTOS", "EYLÜL", "EKİM", "KASIM", "ARALIK", "OCAK"];
        
        const expenses = [
            { name: "GİYİM", amounts: { "AĞUSTOS": 584.50, "EYLÜL": 584.50, "EKİM": 584.50, "KASIM": 584.50 } },
            { name: "DİGİTÜRK", amounts: { "AĞUSTOS": 447.06, "EYLÜL": 447.06, "EKİM": 447.06, "KASIM": 447.06, "ARALIK": 447.06 } },
            { name: "KLAVYE", amounts: { "AĞUSTOS": 606.00, "EYLÜL": 606.00 } },
            { name: "AYAKKABI", amounts: { "AĞUSTOS": 1951.33, "EYLÜL": 1951.33, "EKİM": 1951.33, "KASIM": 1951.33, "ARALIK": 1951.33 } },
            { name: "trendyol", amounts: { "AĞUSTOS": 1597.97, "EYLÜL": 332.45, "EKİM": 332.45 } },
            { name: "kığılı", amounts: { "AĞUSTOS": 2462.45, "EYLÜL": 2462.45, "EKİM": 2462.45, "KASIM": 2462.45, "ARALIK": 2462.45, "OCAK": 2462.45 } },
            { name: "kulaklık", amounts: { "AĞUSTOS": 1074.50, "EYLÜL": 1074.50 } },
            { name: "TELEVİZYON", amounts: { "AĞUSTOS": 944.13, "EYLÜL": 13500.00 } }
        ];

        let selectedMonth = "AĞUSTOS";

        function formatTL(value) {
            return value.toLocaleString('tr-TR', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + " TL";
        }

        function init() {
            // Genel Toplam Hesapla
            let grandTotal = 0;
            expenses.forEach(item => {
                Object.values(item.amounts).forEach(val => grandTotal += val);
            });
            document.getElementById('grandTotal').innerText = formatTL(grandTotal);

            // Ay Butonlarını Oluştur
            const monthSelector = document.getElementById('monthSelector');
            months.forEach(month => {
                const btn = document.createElement('button');
                btn.className = `month-btn ${month === selectedMonth ? 'active' : ''}`;
                btn.innerText = month;
                btn.onclick = () => selectMonth(month);
                monthSelector.appendChild(btn);
            });

            renderList();
        }

        function selectMonth(month) {
            selectedMonth = month;
            document.querySelectorAll('.month-btn').forEach(btn => {
                btn.classList.toggle('active', btn.innerText === month);
            });
            renderList();
        }

        function renderList() {
            const listContainer = document.getElementById('expenseList');
            listContainer.innerHTML = '';
            let monthTotal = 0;

            expenses.forEach(item => {
                if (item.amounts[selectedMonth]) {
                    const amount = item.amounts[selectedMonth];
                    monthTotal += amount;

                    const row = document.createElement('div');
                    row.className = 'list-item';
                    row.innerHTML = `
                        <span class="item-name">${item.name}</span>
                        <span class="item-amount">${formatTL(amount)}</span>
                    `;
                    listContainer.appendChild(row);
                }
            });

            document.getElementById('monthTotal').innerText = formatTL(monthTotal);
        }

        init();
    </script>
